% Figure: guest copy progress in a real acceptance run.
% Columns of data/copy_progress.dat: seconds since copy start, MiB copied.
% The migration instant and the run identity come from data/progress_meta.tex,
% which paper/extract-progress.py regenerates together with the curve, so the
% figure and its caption cannot drift from the measurement.
\begin{tikzpicture}
\begin{axis}[
  width=\columnwidth, height=5.0cm,
  xlabel={time since copy start (s)},
  ylabel={copied (MiB)},
  xmin=0, xmax=\ProgressXMax, ymin=0, ymax=140,
  grid=major, grid style={gray!22},
  tick label style={font=\scriptsize},
  label style={font=\scriptsize},
  legend style={font=\tiny, at={(0.02,0.42)}, anchor=north west, draw=gray!40,
                fill=white, fill opacity=0.85, text opacity=1},
]
  \addplot[very thick, blue!70!black] table[x index=0, y index=1]
    {data/copy_progress.dat};
  \addlegendentry{guest \texttt{dd} progress}
  \addplot[red, dashed, very thick] coordinates
    {(\ProgressMigPoint,0) (\ProgressMigPoint,140)};
  \addlegendentry{migration at $t=\ProgressMigPoint$\,s}
  \addplot[only marks, mark=*, mark size=1.8pt, red]
    coordinates {(\ProgressMigPoint,\ProgressMigMiB)};
\end{axis}
\end{tikzpicture}
